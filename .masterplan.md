#### Example: Strategy Base Class
```python
from abc import ABC, abstractmethod

class Strategy(ABC):
    def __init__(self, params: dict):
        self.params = params
        
    @abstractmethod
    def generate_signal(self, data) -> Signal:
        """Generate buy/sell signal from market data"""
        pass
        
    @abstractmethod
    def calculate_position_size(self, account, signal) -> int:
        """Calculate position size based on risk"""
        pass
        
    def on_bar(self, bar):
        """Called on each new bar"""
        signal = self.generate_signal(bar)
        if signal:
            size = self.calculate_position_size(account, signal)
            return Order(signal.action, size)
```

```python
class ORBStrategy(Strategy):
    def __init__(self):
        self.or_high = None
        self.or_low = None
        self.or_complete = False
        
    def generate_signal(self, bar):
        # Track opening range (9:30-9:45)
        if not self.or_complete:
            self.track_opening_range(bar)
            return None
            
        # Check for breakout
        if bar.close > self.or_high:
            return Signal("BUY", bar.close)
        elif bar.close < self.or_low:
            return Signal("SELL", bar.close)
```

#### Example: Ollama Integration
```python
import requests

class OllamaService:
    def __init__(self, base_url="http://localhost:11434"):
        self.base_url = base_url
        self.model = "llama3.1:70b"
        
    def analyze_chart(self, symbol: str, data: dict) -> str:
        prompt = f"""
        Analyze this chart pattern for {symbol}:
        
        Recent bars: {data['bars']}
        Volume: {data['volume']}
        Indicators: {data['indicators']}
        
        Provide technical analysis and potential setups.
        """
        
        response = requests.post(
            f"{self.base_url}/api/generate",
            json={"model": self.model, "prompt": prompt}
        )
        return response.json()["response"]
```
